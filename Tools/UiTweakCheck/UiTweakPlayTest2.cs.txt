using System;
using System.Collections;
using System.Collections.Generic;
using System.IO;
using System.Reflection;
using System.Text;
using Newtonsoft.Json;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.UI;

[InitializeOnLoad]
public static class UiTweakPlayTest
{
    // Where the copied project goes and the report and screenshots are written.
    static readonly string Dir = Path.Combine(Path.GetTempPath(), "UiTweakCheck") + Path.DirectorySeparatorChar;
    static readonly string Lap = Dir + @"testproj\Death Moon.lap";
    static readonly string Shots = Dir + @"render2\";
    static StringBuilder Log = new StringBuilder();
    static IEnumerator Script;
    static double Started;
    static double WaitUntil;

    static UiTweakPlayTest()
    {
        if (SessionState.GetBool("UiTweakTest", false)) EditorApplication.update += Tick;
    }

    public static void Run()
    {
        SessionState.SetBool("UiTweakTest", true);
        Directory.CreateDirectory(Shots);
        foreach (string F in Directory.GetFiles(Shots)) File.Delete(F);
        File.WriteAllText(Dir + "playtest2.txt", "started\n");
        EditorSceneManager.OpenScene("Assets/_Scenes/LimTuner.unity", OpenSceneMode.Single);
        EditorApplication.update += Tick;
        EditorApplication.isPlaying = true;
    }

    static void Tick()
    {
        if (!EditorApplication.isPlaying) return;
        if (Script == null)
        {
            Started = EditorApplication.timeSinceStartup;
            Application.logMessageReceived += (Msg, Stack, Type) =>
            {
                if (Type == LogType.Exception || Type == LogType.Error || (Type == LogType.Warning && Msg.Contains("UiTweak")))
                    Log.AppendLine("[" + Type + "] " + Msg + "\n" + Stack);
            };
            Script = Steps();
        }
        if (EditorApplication.timeSinceStartup - Started > 300) { Note("TIMEOUT"); Finish(); return; }
        if (EditorApplication.timeSinceStartup < WaitUntil) return;
        try
        {
            if (!Script.MoveNext()) Finish();
        }
        catch (Exception E) { Note("SCRIPT EXCEPTION " + E); Finish(); }
    }

    static void Finish()
    {
        File.WriteAllText(Dir + "playtest2.txt", Log.ToString());
        SessionState.SetBool("UiTweakTest", false);
        EditorApplication.update -= Tick;
        EditorApplication.Exit(0);
    }

    static void Note(string S)
    {
        Log.AppendLine(S);
        File.WriteAllText(Dir + "playtest2.txt", Log.ToString());
    }

    static object Wait(double Seconds) { WaitUntil = EditorApplication.timeSinceStartup + Seconds; return null; }

    static LimTunerManager T { get { return LimTunerManager.Instance; } }

    static IEnumerator Steps()
    {
        yield return Wait(2);
        LimProjectManager Pm = UnityEngine.Object.FindObjectOfType<LimProjectManager>();
        LimProjectManager.CurrentProject = JsonConvert.DeserializeObject<Lanotalium.Project.LanotaliumProject>(File.ReadAllText(Lap));
        LimProjectManager.LapPath = Lap;
        MethodInfo Load = typeof(LimProjectManager).GetMethod("LoadCurrentProject", BindingFlags.NonPublic | BindingFlags.Instance);
        Pm.StartCoroutine((IEnumerator)Load.Invoke(Pm, null));
        double Deadline = EditorApplication.timeSinceStartup + 60;
        while (T == null || !T.isInitialized || T.MediaPlayerManager == null)
        {
            if (EditorApplication.timeSinceStartup > Deadline) { Note("project never loaded"); yield break; }
            yield return null;
        }
        yield return Wait(1);
        LimSystem.Preferences.NoteEffects = LimSystem.Preferences.ComboCounter = LimSystem.Preferences.JudgeLineOrnaments = LimSystem.Preferences.LanotaHeader = true;
        LimSystem.Preferences.FlickArrows = LimSystem.Preferences.BackgroundWave = true;
        LimSystem.Preferences.ShowScore = true;
        LimSystem.Preferences.CustomTunerSkin = "";
        Note("prefs arrows=" + LimSystem.Preferences.FlickArrows + " wave=" + LimSystem.Preferences.BackgroundWave + " score=" + LimSystem.Preferences.ShowScore);
        LimMediaPlayerManager M = T.MediaPlayerManager;

        // The flick out at 31.875 on its way in, and when it is hit.
        M.IsPlaying = false;
        M.Time = 31.35f;
        yield return Wait(0.4);
        Capture("flick_approach_a");
        M.Time = 31.6f;
        yield return Wait(0.25);
        Capture("flick_approach_b");
        yield return Wait(0.25);
        Capture("flick_approach_b_later");
        CountArrows();

        // Taps with the wave and the specks playing.
        M.Time = 13.9f;
        yield return null;
        M.IsPlaying = true;
        foreach (object Step in ShootAt(new[] { 14.3f, 15.1f })) yield return Step;
        M.IsPlaying = false;
        yield return Wait(0.3);
        for (int i = 0; i < 3; ++i)
        {
            M.Time = 20f + i * 0.125f;
            yield return Wait(0.1);
            Capture("wave_phase_" + i);
        }

        // Full screen with the header, the score and a long name.
        LimDisplayManager Display = UnityEngine.Object.FindObjectOfType<LimDisplayManager>();
        Camera Cam = UnityEngine.Object.FindObjectOfType<LimTunerWindowManager>().TunerCamera;
        RenderTexture Rt = new RenderTexture(1280, 720, 24);
        M.Time = 30f;
        Display.FullScreenTuner = true;
        Cam.targetTexture = Rt;
        yield return Wait(0.4);
        Read(Rt, "full_score");
        LimTunerHeadManager Head = UnityEngine.Object.FindObjectOfType<LimTunerHeadManager>();
        string OldName = Head.ChartName.text;
        Head.ChartName.text = "A Very Long Chart Title That Goes On And On Forever";
        Head.DesignerInputField.text = "Designer";
        LimSystem.Preferences.ShowScore = false;
        yield return Wait(0.3);
        Read(Rt, "full_designer_longname");
        Head.DesignerInputField.text = "Somebody With A Really Long Designer Name";
        yield return Wait(0.3);
        Read(Rt, "full_designer_long");
        Head.ChartName.text = OldName;
        Head.DesignerInputField.text = "Designer";
        LimSystem.Preferences.ShowScore = true;
        M.IsPlaying = true;
        yield return Wait(1.0);
        M.IsPlaying = false;
        yield return Wait(0.2);
        Read(Rt, "full_score_playing");
        Note("score text: " + FindText(Head.transform, "Value"));
        Cam.targetTexture = null;
        Display.FullScreenTuner = false;
        yield return Wait(0.3);

        // The UiTweak menu.
        Camera Main = Display.MainCamera;
        RenderTexture Wide = new RenderTexture(1600, 900, 24);
        Main.targetTexture = Wide;
        LimTopMenuManager Menu = UnityEngine.Object.FindObjectOfType<LimTopMenuManager>();
        Transform Tab = Menu.transform.Find("UiTweak/ChartConvert");
        Note("tab found=" + (Tab != null));
        Tab.GetComponent<Button>().onClick.Invoke();
        yield return Wait(0.3);
        Read(Wide, "menu_open");
        Transform Panel = Menu.transform.Find("UiTweak/UiTweakPanel");
        Transform Arrows = Panel.Find("UiTweak_FlickArrows");
        Arrows.GetComponent<Button>().onClick.Invoke();
        yield return null;
        Note("after click arrows: pref=" + LimSystem.Preferences.FlickArrows + " check='" + Arrows.Find("Check").GetComponent<Text>().text + "'");
        yield return null;
        CountArrows();
        Arrows.GetComponent<Button>().onClick.Invoke();
        Panel.Find("TunerSkin").GetComponent<Button>().onClick.Invoke();
        yield return Wait(0.3);
        Read(Wide, "menu_skins");
        Transform Skins = Panel.Find("SkinPanel");
        Skins.Find("Astra").GetComponent<Button>().onClick.Invoke();
        yield return null;
        Note("after Astra: pref='" + LimSystem.Preferences.CustomTunerSkin + "' check='" + Skins.Find("Astra/Check").GetComponent<Text>().text + "'");
        Menu.GetComponentInChildren<LimLanguageManager>(true);
        Main.targetTexture = null;
        Capture("skin_astra_from_menu");
        LimSystem.Preferences.CustomTunerSkin = "";
        Note("done");
    }

    static string FindText(Transform Root, string Name)
    {
        foreach (Text X in Root.GetComponentsInChildren<Text>(true)) if (X.name == Name) return X.text + " active=" + X.gameObject.activeInHierarchy;
        return "none";
    }

    static void CountArrows()
    {
        int Flicks = 0, Arrows = 0;
        foreach (Lanotalium.Chart.LanotaTapNote Note in T.TapNoteManager.TapNote)
        {
            if (Note.Type != 2 && Note.Type != 3) continue;
            ++Flicks;
            if (Note.TapNoteGameObject.transform.Find("FlickArrow") != null) ++Arrows;
        }
        UiTweakPlayTest.Note("flicks=" + Flicks + " with arrows=" + Arrows);
    }

    static IEnumerable<object> ShootAt(float[] At)
    {
        foreach (float Time in At)
        {
            double Deadline = EditorApplication.timeSinceStartup + 10;
            while (T.ChartTime < Time && EditorApplication.timeSinceStartup < Deadline) yield return null;
            Capture("t" + T.ChartTime.ToString("00.000"));
        }
    }

    static void Capture(string Name)
    {
        Camera Cam = UnityEngine.Object.FindObjectOfType<LimTunerWindowManager>().TunerCamera;
        RenderTexture Rt = RenderTexture.GetTemporary(1280, 720, 24);
        RenderTexture Old = Cam.targetTexture;
        Cam.targetTexture = Rt;
        Cam.Render();
        Cam.targetTexture = Old;
        Read(Rt, Name);
        RenderTexture.ReleaseTemporary(Rt);
    }

    static void Read(RenderTexture Rt, string Name)
    {
        RenderTexture Before = RenderTexture.active;
        RenderTexture.active = Rt;
        Texture2D Tex = new Texture2D(Rt.width, Rt.height, TextureFormat.RGB24, false);
        Tex.ReadPixels(new Rect(0, 0, Rt.width, Rt.height), 0, 0);
        Tex.Apply();
        RenderTexture.active = Before;
        File.WriteAllBytes(Shots + Name + ".png", Tex.EncodeToPNG());
        UnityEngine.Object.DestroyImmediate(Tex);
        Note(Name + ": chart=" + T.ChartTime.ToString("0.000") + " playing=" + T.MediaPlayerManager.IsPlaying);
    }
}
